el-framework

A standalone CSS framework with 12 themes and 100+ elements. Works in HTML, TypeScript, and Python/Tkinter.

Install

# HTML
<link rel="stylesheet" href="el-framework/tokens.css">
<link rel="stylesheet" href="el-framework/themes.css">
<link rel="stylesheet" href="el-framework/base.css">
<link rel="stylesheet" href="el-framework/elements.css">

# npm
npm install el-framework

# Python (Tkinter)
from el_framework import get_theme, ElWidget
theme = get_theme('polaroid')
btn = ElWidget(parent, 'btn', theme=theme, text='OK')

# TypeScript
import 'el-framework';
const pill = new ElPill(document.body);
